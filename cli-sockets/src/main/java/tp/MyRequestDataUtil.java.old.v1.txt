package tp;

public class MyRequestDataUtil {
    public static byte[] lastNewsMessageData(){
        MyDataMessageV1 dataMessage = new MyDataMessageV1("last_news");
        return dataMessage.get_main_data();
    }

    public static byte[] averageDoubleMessage(double[] values){
        MyDataMessageV1 dataMessage = new MyDataMessageV1("average_double",values.length, 8, MyDataUtilV1.bytesFromDoubleArray(values));
        return dataMessage.get_full_data();
    }

}
